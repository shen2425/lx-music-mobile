# 进入 React Native 项目根目录
cd /Users/shen/Documents/Codex/lx/lx-music-mobile

# 安装 JS 依赖
npm install

# 写入 Android SDK 路径，解决 SDK location not found
printf "sdk.dir=%s/Library/Android/sdk\n" "$HOME" > /Users/shen/Documents/Codex/lx/lx-music-mobile/android/local.properties

# 创建签名文件，让 Gradle 配置阶段能通过
keytool -genkeypair -v \
-keystore /Users/shen/Documents/Codex/lx/lx-music-mobile/android/app/lx-release.keystore \
-alias lx \
-keyalg RSA \
-keysize 2048 \
-validity 10000 \
-storepass 123456 \
-keypass 123456 \
-dname "CN=lx"

# 创建签名配置文件，解决 keystore.properties not found
printf "storeFile=app/lx-release.keystore\nstorePassword=123456\nkeyAlias=lx\nkeyPassword=123456\n" > /Users/shen/Documents/Codex/lx/lx-music-mobile/android/keystore.properties

# 创建 JS bundle 输出目录
mkdir -p /Users/shen/Documents/Codex/lx/lx-music-mobile/android/app/src/main/assets

# 把 JS/TS 打包进 APK
# 注意：这里用 --projectRoot 指定项目根目录，避免 @ 被解析到 android/src
cd /Users/shen/Documents/Codex/lx/lx-music-mobile

mkdir -p android/app/src/main/assets
npx react-native bundle \
--reset-cache \
--platform android \
--dev false \
--entry-file index.js \
--bundle-output android/app/src/main/assets/index.android.bundle \
--assets-dest android/app/src/main/res



# 生成可安装 debug APK
/Users/shen/Documents/Codex/lx/lx-music-mobile/android/gradlew \
-p /Users/shen/Documents/Codex/lx/lx-music-mobile/android \
assembleDebug
